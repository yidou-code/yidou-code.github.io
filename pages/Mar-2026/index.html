# Technical Difficulties Overcome

# Node.js

  1. Learn how to separate the CSS and JS in HTML files
  2. learn how to extract CSS and JS from HTML and write them to separate files
  3. Deepen the understanding and application of the "replace" and "exec" method
// 引入文件处理和路径处理模块
const fs = require('fs')
const path = require('path')

// 定义正则：匹配<style>标签及内部所有内容（[\s\S]匹配所有字符，包括换行）
const regStyle = /<style>[\s\S]*<\/style>/
// 定义正则：匹配<script>标签及内部所有内容
const regScript = /<script>[\s\S]*<\/script>/

// 读取目标 HTML 文件（绝对路径拼接，避免相对路径问题）
fs.readFile(
  path.join(__dirname, './素材/index.html'),
  'utf-8',
  (err, data) => {
    if (err) {
      console.log('读取 HTML 文件失败：', err.message)
      return // 终止后续执行
    }
    // 读取成功后，依次处理 CSS、JS、HTML
    resolveCSS(data)
    resolveJS(data)
    resolveHTML(data)
  }
)

/**
 * 处理 CSS：提取内嵌样式并写入独立 CSS 文件
 * @param {string} htmlStr - 完整的 HTML 字符串
 */
function resolveCSS(htmlStr) {
  // 执行正则匹配，获取 <style> 及内部内容
  const matchResult = regStyle.exec(htmlStr)
  // 容错：如果没匹配到 CSS 内容，直接返回
  if (!matchResult) {
    console.log('⚠️ 未匹配到任何 CSS 样式内容')
    return
  }
  // 剔除 <style> 标签，只保留纯 CSS 内容
  const cssContent = matchResult[0]
    .replace('<style>', '')
    .replace('</style>', '')
  
  // 写入 CSS 文件
  fs.writeFile(
    path.join(__dirname, './素材/index.css'),
    cssContent,
    (err) => {
      if (err) {
        console.log('写入 CSS 文件失败：', err.message)
        return
      }
      console.log('✅ CSS 文件写入成功！')
    }
  )
}

/**
 * 处理 JS：提取内嵌脚本并写入独立 JS 文件
 * @param {string} htmlStr - 完整的 HTML 字符串
 */
function resolveJS(htmlStr) {
  const matchResult = regScript.exec(htmlStr)
  if (!matchResult) {
    console.log('⚠️ 未匹配到任何 JS 脚本内容')
    return
  }
  // 剔除 <script> 标签，只保留纯 JS 内容
  const jsContent = matchResult[0]
    .replace('<script>', '')
    .replace('</script>', '')
  
  // 写入 JS 文件
  fs.writeFile(
    path.join(__dirname, './素材/index.js'),
    jsContent,
    (err) => {
      if (err) {
        console.log('写入 JS 文件失败：', err.message)
        return
      }
      console.log('✅ JS 文件写入成功！')
    }
  )
}

/**
 * 处理 HTML：替换内嵌的 style/script 为外部引入链接
 * @param {string} htmlStr - 完整的 HTML 字符串
 */
function resolveHTML(htmlStr) {
  // 替换 style 标签为 link 引入 CSS，替换 script 标签为引入 JS
  const newHtmlContent = htmlStr
    .replace(regStyle, '<link rel="stylesheet" href="index.css">')
    .replace(regScript, '<script src="index.js"></script>')
  
  // 写入修改后的 HTML 文件（覆盖原文件）
  fs.writeFile(
    path.join(__dirname, './素材/index.html'), // 修正原代码的 _dirname_ 笔误
    newHtmlContent,
    (err) => {
      if (err) {
        console.log('写入 HTML 文件失败：', err.message)
        return
      }
      console.log('✅ HTML 文件修改成功！')
    }
  )
}
